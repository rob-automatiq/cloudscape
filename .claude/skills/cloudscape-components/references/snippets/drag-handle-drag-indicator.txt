// sentiment.jsx

import React from 'react';
import { ColumnLayout, Icon } from '@cloudscape-design/components';
import clsx from 'clsx';
import './styles.scss';

export function AllDragHandles() {
  return (
    <ColumnLayout columns={4}>
      <DragHandleAreaResize />
      <DragHandleVerticalResize />
      <DragHandleHorizontalResize />
      <DragHandleDragIndicator />
    </ColumnLayout>
  );
}

export function DragHandleAreaResize() {
  return <InternalDragHandle variant="resize-area" ariaLabel="Area resize handle" />;
}

export function DragHandleVerticalResize() {
  return <InternalDragHandle variant="resize-vertical" ariaLabel="Vertical resize handle" />;
}

export function DragHandleHorizontalResize() {
  return <InternalDragHandle variant="resize-horizontal" ariaLabel="Horizontal resize handle" />;
}

export function DragHandleDragIndicator() {
  return <InternalDragHandle variant="drag-indicator" ariaLabel="Drag handle" />;
}

function InternalDragHandle({ variant = 'drag-indicator', ariaLabel }) {
  const iconProps = (() => {
    switch (variant) {
      case 'drag-indicator':
        return { name: 'drag-indicator' };
      case 'resize-area':
        return { name: 'resize-area' };
      case 'resize-horizontal':
        return { svg: <ResizeIcon variant="horizontal" /> };
      case 'resize-vertical':
        return { svg: <ResizeIcon variant="vertical" /> };
    }
  })();

  return (
    <div
      role="button"
      tabIndex={0}
      className={clsx('drag-handle-handle', `drag-handle-handle-${variant}`)}
      aria-label={ariaLabel}
    >
      <Icon {...iconProps} />
    </div>
  );
}

function ResizeIcon({ variant }) {
  return (
    <svg
      focusable={false}
      className={clsx('drag-handle-resize-icon', `drag-handle-resize-icon-${variant}`)}
      xmlns="http://www.w3.org/2000/svg"
      width="16"
      height="16"
      viewBox="0 0 16 16"
      aria-hidden={true}
    >
      <line strokeWidth="2" x1="2" y1="5" x2="14" y2="5" />
      <line strokeWidth="2" x1="14" y1="10" x2="2" y2="10" />
    </svg>
  );
}
