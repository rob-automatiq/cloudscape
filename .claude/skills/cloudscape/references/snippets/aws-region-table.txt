const columnDefinitions = [
  {
    id: 'id',
    header: 'Resource ID',
    cell: (item: ResourceDefinition) => item.id,
    isRowHeader: true,
    width: 185,
  },
  {
    id: 'state',
    cell: (item: ResourceDefinition) => (
      <StatusIndicator type={item.state.toLocaleLowerCase() === 'inactive' ? 'stopped' : 'success'}>
        {item.state}
      </StatusIndicator>
    ),
    header: 'Status',
    minWidth: 110,
    width: 110,
  },
  {
    id: 'region',
    header: 'Region',
    cell: (item: ResourceDefinition) => formatReadOnlyRegion(item.regionCode),
    minWidth: 210,
    width: 285,
  },
  {
    id: 'date',
    header: 'Last modified',
    cell: (item: ResourceDefinition) => formatDate(item.lastModified),
    ariaLabel: ({ sorted, descending }: TableProps.LabelData) =>
      `Last modified, ${sorted ? `sorted ${descending ? 'descending' : 'ascending'}` : 'not sorted'}.`,
    sortingComparator: (a: ResourceDefinition, b: ResourceDefinition) =>
      a.lastModified.valueOf() - b.lastModified.valueOf(),
  },
];

function formatDate(date: Date) {
  const dateFormatter = new Intl.DateTimeFormat('en-US', { dateStyle: 'long' });
  const timeFormatter = new Intl.DateTimeFormat('en-US', { timeStyle: 'short', hour12: false });
  return `${dateFormatter.format(date)}, ${timeFormatter.format(date)}`;
}

const RegionTable = () => {
  const { items, collectionProps } = useCollection(tableItems, {
    sorting: { defaultState: { sortingColumn: columnDefinitions[3], isDescending: true } },
  });
  return (
    <Table
      header={
        <Header headingTagOverride="h3" counter={`(${tableItems.length})`}>
          Resources
        </Header>
      }
      columnDefinitions={columnDefinitions}
      items={items}
      resizableColumns={true}
      {...collectionProps}
    ></Table>
  );
};

export default RegionTable;
