function NodeButton({
  className,
  style,
  variant = 'default',
  onClick,
  children,
}: {
  className: string;
  style: CSSProperties;
  variant?: 'accent' | 'default';
  onClick?: () => void;
  children: React.ReactNode;
}) {
  const [hovered, setHovered] = useState(false);
  const hoverStyle = variant === 'accent' ? styles.nodeAccentHover : styles.nodeDefaultHover;
  return (
    <button
      className={className}
      style={{ ...style, ...(hovered ? hoverStyle : {}) }}
      type="button"
      onClick={onClick}
      onMouseEnter={() => setHovered(true)}
      onMouseLeave={() => setHovered(false)}
    >
      {children}
    </button>
  );
}

function TagButton({
  className,
  style,
  highlighted,
  onClick,
  children,
}: {
  className: string;
  style: CSSProperties;
  highlighted: boolean;
  onClick?: () => void;
  children: React.ReactNode;
}) {
  const [hovered, setHovered] = useState(false);
  return (
    <button
      className={className}
      style={{ ...style, ...(hovered && highlighted ? styles.tagHighlightedHover : {}) }}
      type="button"
      onClick={onClick}
      onMouseEnter={() => setHovered(true)}
      onMouseLeave={() => setHovered(false)}
    >
      {children}
    </button>
  );
}

export default function InteractiveOverview() {
  const { filterText, setFilterText, categoryCounts } = useHubFilter();

  // Labels + order come from the shared taxonomy; counts come from live tag data
  // via context. When no counts are provided (e.g. standalone snippet preview),
  // the count badges are omitted rather than showing stale numbers.
  const flowNodes: FlowNode[] = DIAGRAM_FLOW_STAGES.map((stage, index) => ({
    label: stage.label,
    count: categoryCounts[stage.label],
    variant: index === 0 ? 'accent' : 'default',
  }));

  const tags: Tag[] = [
    { label: DIAGRAM_CHAT_STAGE.label, count: categoryCounts[DIAGRAM_CHAT_STAGE.label], highlighted: true },
    { label: 'Image' },
    { label: 'Text' },
  ];

  const handleFilterClick = (label: string) => {
    // Toggle: clicking the same label again clears the filter
    setFilterText(filterText === label ? '' : label);
  };

  return (
    <div className="container" style={styles.container}>
      <svg style={styles.borderSvg}>
        <rect
          rx={tokens.borderRadiusContainer}
          ry={tokens.borderRadiusContainer}
          fill="none"
          stroke={tokens.colorBorderControlDefault}
          strokeWidth="1"
          strokeDasharray="8 6"
          strokeDashoffset="4"
          style={{ x: 1, y: 1, width: 'calc(100% - 2px)', height: 'calc(100% - 2px)' }}
        />
      </svg>
      <span className="title" style={styles.title}>
        <Box variant="h4">Pattern abstraction</Box>
      </span>

      <div className="scrollable" style={styles.scrollable}>
        <div style={{ margin: '0 auto' }}>
          <div className="flow" style={styles.flow}>
            {/* Ingress node (outside the group) */}
            <Box padding={{ top: 'l' }}>
              <NodeButton
                className="node node--accent"
                variant="accent"
                style={{
                  ...styles.node,
                  ...styles.nodeAccent,
                  ...(filterText === flowNodes[0].label ? styles.nodeAccentHover : {}),
                }}
                onClick={() => handleFilterClick(flowNodes[0].label)}
              >
                <span>{flowNodes[0].label}</span>
                {flowNodes[0].count !== undefined && (
                  <span className="nodeCount" style={styles.nodeCount}>
                    {flowNodes[0].count}
                  </span>
                )}
              </NodeButton>
            </Box>

            <div className="connectorWrapper" style={styles.connectorWrapper}>
              <div className="connector" style={styles.connector}>
                <div className="connectorLine" style={styles.connectorLine} />
                <div className="connectorArrow" style={styles.connectorArrow} />
              </div>
            </div>

            {/* Grouped nodes */}
            <div className="group" style={styles.group}>
              <div className="groupNodes" style={styles.groupNodes}>
                {flowNodes.slice(1).map((node, idx) => (
                  <div className="segment" style={styles.segment} key={node.label}>
                    <NodeButton
                      className="node node--default"
                      style={{
                        ...styles.node,
                        ...styles.nodeDefault,
                        ...(filterText === node.label ? styles.nodeDefaultHover : {}),
                      }}
                      onClick={() => handleFilterClick(node.label)}
                    >
                      <span>{node.label}</span>
                      {node.count !== undefined && (
                        <span
                          className="nodeCount nodeCount--bordered"
                          style={{ ...styles.nodeCount, ...styles.nodeCountBordered }}
                        >
                          {node.count}
                        </span>
                      )}
                    </NodeButton>
                    {idx < flowNodes.length - 2 && (
                      <div className="connector" style={styles.connector}>
                        <div className="connectorLine" style={styles.connectorLine} />
                        <div className="connectorArrow" style={styles.connectorArrow} />
                      </div>
                    )}
                  </div>
                ))}
              </div>

              {/* Tags row */}
              <div className="tags" style={styles.tags}>
                {tags.map(tag => {
                  const isInteractive = tag.count !== undefined;
                  const isActive = filterText === tag.label;
                  const tagStyle = {
                    ...styles.tag,
                    ...(tag.highlighted || isActive ? styles.tagHighlighted : {}),
                    ...(isInteractive ? styles.tagInteractive : {}),
                  };
                  const tagClassName = [
                    'tag',
                    tag.highlighted || isActive ? 'tag--highlighted' : '',
                    isInteractive ? 'tag--interactive' : '',
                  ]
                    .filter(Boolean)
                    .join(' ');

                  if (isInteractive) {
                    return (
                      <TagButton
                        className={tagClassName}
                        style={tagStyle}
                        highlighted={!!tag.highlighted || isActive}
                        key={tag.label}
                        onClick={() => handleFilterClick(tag.label)}
                      >
                        <span>{tag.label}</span>
                        <span className="tagCount" style={styles.tagCount}>
                          {tag.count}
                        </span>
                      </TagButton>
                    );
                  }

                  return (
                    <span className={tagClassName} style={tagStyle} key={tag.label}>
                      <span>{tag.label}</span>
                    </span>
                  );
                })}
                <span className="annotation" style={styles.annotation}>
                  interaction paradigm,
                </span>
                <span className="annotation" style={styles.annotation}>
                  model modality
                </span>
              </div>
            </div>
          </div>
        </div>
      </div>

      <span className="context" style={styles.context}>
        Context
      </span>
    </div>
  );
}
