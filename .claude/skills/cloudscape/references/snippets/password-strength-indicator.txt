'use client';
import { FormField, Input, SpaceBetween } from '@cloudscape-design/components';
import { PasswordStrengthIndicator } from '@amzn/password-strength-indicator';
import { useState } from 'react';

export default function PasswordStrengthIndicatorExample() {
  const [passwordFieldTouched, setPasswordFieldTouched] = useState(false);
  const [confirmFieldTouched, setConfirmFieldTouched] = useState(false);
  const [confirmValue, setConfirmValue] = useState('');
  const [passwordValue, setPasswordValue] = useState('');
  const passwordErrorMessage =
    passwordFieldTouched && passwordValue.length < 5 ? 'Enter at least five characters.' : undefined;
  const passwordIdPrefix = 'password-one';
  const strengthIndicatorVisible = passwordValue && passwordFieldTouched && !passwordErrorMessage;

  return (
    <SpaceBetween size="l">
      <FormField
        label="Password"
        errorText={passwordErrorMessage}
        constraintText={
          <span id={`${passwordIdPrefix}-constraint`}>Password must be at least five characters long.</span>
        }
      >
        <SpaceBetween size="xs">
          <Input
            type="password"
            value={passwordValue}
            onChange={({ detail }) => setPasswordValue(detail.value)}
            onBlur={() => setPasswordFieldTouched(true)}
            ariaDescribedby={strengthIndicatorVisible && `${passwordIdPrefix}-constraint ${passwordIdPrefix}-indicator`}
          />
          {strengthIndicatorVisible && (
            <PasswordStrengthIndicator id={`${passwordIdPrefix}-indicator`} value={passwordValue} />
          )}
        </SpaceBetween>
      </FormField>
      <FormField
        label="Confirm Password"
        errorText={
          confirmFieldTouched && confirmValue !== passwordValue ? 'Enter the same password to confirm.' : undefined
        }
      >
        <Input
          type="password"
          value={confirmValue}
          onChange={({ detail }) => setConfirmValue(detail.value)}
          onBlur={() => setConfirmFieldTouched(true)}
        />
      </FormField>
    </SpaceBetween>
  );
}
