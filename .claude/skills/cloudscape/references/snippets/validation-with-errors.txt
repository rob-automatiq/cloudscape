import React from 'react';
import Form from './common-form';

const Snippet = () => {
  const validate = value => {
    if (!value.match('^[a-z0-9-]*$')) {
      return 'The specified value for the instance name is invalid. It must contain only lowercase alphanumeric characters and/or the following: - (hyphen).';
    }
    if (value.length === 0) {
      return 'Instance name is required.';
    }
    if (value.length < 8) {
      return 'Instance name should be longer than 7 characters.';
    }
    if (value.length > 128) {
      return 'Instance name should be shorter than 129 characters.';
    }
    return '';
  };

  return <Form validate={validate} />;
};

export default Snippet;
